        \footnotesize
        \vspace{0.4em}
        \textbf{\color{pcNavy!70}Anode}\\[0.3em]
        \centerline{\normalsize\ch{H2O -> 2 H+ + 2 e- + 1/2 O2}}\\[0.9em]
        \textbf{\color{pcNavy}Cathode}\\[0.3em]
        \centerline{\normalsize\ch{CO2 + 2 H+ + 2 e- -> CO + H2O}}
      \end{pcbanner}

      \vspace{0.4em}

      \begin{pcbanner}{Principe}
        \begin{itemize}\footnotesize
          \item Deux parties : photovoltaïque + électrochimique
          \item Soleil $\rightarrow$ charges électriques (partie PV)
          \item Charges $\rightarrow$ réduction du \ch{CO2} (partie électrochimique)
        \end{itemize}
      \end{pcbanner}

\section{Bilan sur la première année}
\begin{frame}[t]{Bilan : brique photovoltaïque}
  \vspace{0.3em}
  \begin{center}
    {\footnotesize\color{pcNavy!70}Solveur drift-diffusion 1D développé en C, basé sur Van Roosbroeck}
  \end{center}
  \vspace{0.15em}

  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{pcbanner}{Ce que j'ai construit}
        \begin{itemize}\footnotesize
          \item Solveur 1D : Poisson + deux continuités
          \item Absorption, génération, collecte des porteurs
          \item Recombinaisons SRH, Auger, radiatif
          \item Tentatives d'ajout de l'hétérojonction CIGS/CdS/ZnO
        \end{itemize}
      \end{pcbanner}
    \column{0.48\textwidth}
      \begin{pcbanner}{Jalons clés}
        \begin{itemize}\footnotesize
          \item 1\textsuperscript{re} version : Newton stationnaire, adimensionnée
          \item Blocage à la photogénération
          \item Levé par passage en résolution transitoire
        \end{itemize}
      \end{pcbanner}
  \end{columns}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \column{\textwidth}
      \begin{pcbanner}{Validation}
        \begin{itemize}\footnotesize
          \item Cas sans génération : accord avec Farrell et al.
          \item Comparaison avec SCAPS 1D
        \end{itemize}
      \end{pcbanner}
  \end{columns}
\end{frame}

% ---------- BILAN : BRIQUE ÉLECTROCHIMIQUE ----------
\begin{frame}[t]{Bilan : brique électrochimique}
  \vspace{0.3em}
  \begin{center}
    {\footnotesize\color{pcNavy!70}Modèle homogénéisé, de l'échelle du pore au 1D macroscopique}
  \end{center}
  \vspace{0.15em}

  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{pcbanner}{Ce que j'ai construit}
        \begin{itemize}\footnotesize
          \item Modèle 3D local : Nernst--Planck, loi d'Ohm, Butler--Volmer
          \item Réduction 1D par homogénéisation (coefficients effectifs)
        \end{itemize}
      \end{pcbanner}
    \column{0.48\textwidth}
      \begin{pcbanner}{Jalons clés}
        \begin{itemize}\footnotesize
          \item Étude bibliographique (lois de vitesse)
          \item Modèle de Thiele en cas limite (facteur d'efficacité)
          \item Revue des phénomènes influençant l'électrode
        \end{itemize}
      \end{pcbanner}
  \end{columns}

  \vspace{0.3em}

      \end{pcbanner}
  \end{columns}
\end{frame}

          \item Intégrer des phénomènes fins (HER, pH, double couche\dots)
        \end{itemize}
      \end{pcbanner}

          \item Génération optique : modélisation électromagnétique
          \item \textit{Detailed balance} : approche alternative (semi-conducteur)
        \end{itemize}
      \end{pcbanner}
  \end{columns}
